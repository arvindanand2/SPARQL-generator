PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT DISTINCT ?manufacturer ?label WHERE {
  ?manufacturer a iof-core:Manufacturer ;
                rdfs:label ?label ;
                sdk:hasMaterialCapability ?mc .
  ?mc a ?mctype .
  FILTER (?mctype IN (sdk:UrethaneProcessingCapability, sdk:StainlessSteelProcessingCapability, sdk:BronzeProcessingCapability))
  
  FILTER NOT EXISTS {
    ?manufacturer sdk:hasMaterialCapability ?otherMc .
    ?otherMc a ?otherType .
    FILTER (?otherType NOT IN (sdk:UrethaneProcessingCapability, sdk:StainlessSteelProcessingCapability, sdk:BronzeProcessingCapability))
  }
}

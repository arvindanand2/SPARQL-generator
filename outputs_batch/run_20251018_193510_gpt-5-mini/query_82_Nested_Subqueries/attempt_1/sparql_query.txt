PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT ?bucket (COUNT(?company) AS ?companyCount)
WHERE {
  ?company a iof-core:Manufacturer ;
           sdk:hasNumberOfEmployees ?employees .
  BIND(xsd:integer(REPLACE(STR(?employees), ",", "")) AS ?empInt)
  FILTER(?empInt > 250)
  BIND(
    IF(?empInt <= 500, "251-500",
      IF(?empInt <= 1000, "501-1000", ">1000")
    ) AS ?bucket
  )
}
GROUP BY ?bucket
ORDER BY ?bucket

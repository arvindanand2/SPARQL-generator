PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT DISTINCT ?manufacturer ?manufacturerLabel ?cap ?capLabel WHERE {
  ?manufacturer a iof-core:Manufacturer ;
                rdfs:label ?manufacturerLabel ;
                sdk:hasMaterialCapability ?cap .
  ?cap a ?capClass ;
       rdfs:label ?capLabel .
  VALUES ?capClass { sdk:CarbonProcessingCapability sdk:GraphiteProcessingCapability }
}
ORDER BY ?manufacturerLabel

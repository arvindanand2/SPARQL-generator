PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT DISTINCT ?manufacturer ?label
WHERE {
  ?manufacturer a iof-core:Manufacturer ;
                sdk:suppliesToIndustry ?industry ;
                sdk:hasProcessCapability ?cap ;
                sdk:hasNumberOfEmployees ?emp .
  OPTIONAL { ?manufacturer rdfs:label ?label. }

  ?industry a sdk:RenewableEnergyIndustry .

  ?cap a ?capClass .
  ?capClass rdfs:subClassOf* sdk:CoatingCapability .

  FILTER(xsd:integer(REPLACE(STR(?emp), ",", "")) > 200)
}

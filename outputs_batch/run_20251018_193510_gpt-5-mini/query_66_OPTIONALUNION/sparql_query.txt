PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?manufacturer ?manufacturerLabel (GROUP_CONCAT(DISTINCT ?itemLabelFinal; separator=", ") AS ?items)
WHERE {
  ?manufacturer a iof-core:Manufacturer .
  OPTIONAL { ?manufacturer rdfs:label ?manufacturerLabel . }
  OPTIONAL {
    { ?manufacturer sdk:manufactures ?item }
    UNION
    { ?manufacturer sdk:hasCertificate ?item }
    OPTIONAL { ?item rdfs:label ?itemLabel }
    BIND(COALESCE(?itemLabel, STR(?item)) AS ?itemLabelFinal)
  }
}
GROUP BY ?manufacturer ?manufacturerLabel
ORDER BY ?manufacturerLabel

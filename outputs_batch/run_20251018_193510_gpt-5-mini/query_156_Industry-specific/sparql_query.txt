PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX iof-scro: <https://spec.industrialontologies.org/ontology/supplychain/SupplyChain/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?company ?label
WHERE {
  ?company sdk:suppliesToIndustry ?ind .
  ?ind a ?type .
  OPTIONAL { ?company rdfs:label ?label. }
  FILTER (?type IN (sdk:RenewableEnergyIndustry, sdk:UtilitiesIndustry))
}
GROUP BY ?company ?label
HAVING (
  COUNT(DISTINCT ?ind) = 2
  && SUM(IF(?type = sdk:RenewableEnergyIndustry, 1, 0)) >= 1
  && SUM(IF(?type = sdk:UtilitiesIndustry, 1, 0)) >= 1
)
LIMIT 2

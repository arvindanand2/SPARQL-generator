PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
PREFIX iof-scro: <https://spec.industrialontologies.org/ontology/supplychain/SupplyChain/>

SELECT ?org
       (COALESCE(STR(?label), "Unknown") AS ?labelOrUnknown)
       (COALESCE(STR(?numEmp), "Unknown") AS ?numEmployeesOrUnknown)
       (COALESCE(STR(?descText), "Unknown") AS ?descriptionOrUnknown)
       (IF(STRLEN(COALESCE(GROUP_CONCAT(DISTINCT ?procLabel; separator=", "), "")) > 0,
           GROUP_CONCAT(DISTINCT ?procLabel; separator=", "),
           "Unknown") AS ?processCapabilities)
       (IF(STRLEN(COALESCE(GROUP_CONCAT(DISTINCT ?matLabel; separator=", "), "")) > 0,
           GROUP_CONCAT(DISTINCT ?matLabel; separator=", "),
           "Unknown") AS ?materialCapabilities)
WHERE {
  ?org a iof-core:Manufacturer .
  OPTIONAL { ?org rdfs:label ?label. }
  OPTIONAL { ?org sdk:hasNumberOfEmployees ?numEmp. }
  OPTIONAL { ?org iof-core:describedBy ?desc . ?desc iof-scro:hasTextValue ?descText. }
  OPTIONAL { ?org sdk:hasProcessCapability ?procCap . ?procCap rdfs:label ?procLabel. }
  OPTIONAL { ?org sdk:hasMaterialCapability ?matCap . ?matCap rdfs:label ?matLabel. }
}
GROUP BY ?org ?label ?numEmp ?descText

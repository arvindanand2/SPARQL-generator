PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?manufacturer ?label ?naics ?naicsCode ?naicsText
WHERE {
  ?manufacturer a iof-core:Manufacturer ;
                rdfs:label ?label .
  FILTER(CONTAINS(LCASE(STR(?label)), "pipe"))
  OPTIONAL {
    ?manufacturer sdk:hasPrimaryNAICSClassifier ?naics .
    OPTIONAL { ?naics sdk:hasNAICSCodeValue ?naicsCode . }
    OPTIONAL { ?naics sdk:hasNAICSTextValue ?naicsText . }
  }
}

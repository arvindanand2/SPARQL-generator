PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?industry ?industryLabel ?count WHERE {
  {
    SELECT ?industry (COUNT(DISTINCT ?manufacturer) AS ?count) WHERE {
      ?manufacturer a iof-core:Manufacturer ;
                    sdk:suppliesToIndustry ?industry .
    } GROUP BY ?industry
  }
  {
    SELECT (MAX(?c) AS ?maxCount) WHERE {
      {
        SELECT ?i (COUNT(DISTINCT ?m) AS ?c) WHERE {
          ?m a iof-core:Manufacturer ;
             sdk:suppliesToIndustry ?i .
        } GROUP BY ?i
      }
    }
  }
  FILTER(?count = ?maxCount)
  OPTIONAL { ?industry rdfs:label ?industryLabel }
}
ORDER BY ?industryLabel

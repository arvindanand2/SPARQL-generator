PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
PREFIX iof-scro: <https://spec.industrialontologies.org/ontology/supplychain/SupplyChain/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
PREFIX bfo: <http://purl.obolibrary.org/obo/>

SELECT ?industry ?industryLabel (COUNT(DISTINCT ?org) AS ?supplierCount)
WHERE {
  ?org a iof-core:Manufacturer ;
       sdk:suppliesToIndustry ?industry ;
       sdk:manufactures ?prod .
  ?prod rdfs:label ?prodLabel .
  ?industry rdfs:label ?industryLabel .
  FILTER(CONTAINS(LCASE(STR(?prodLabel)), "pipe"))
}
GROUP BY ?industry ?industryLabel
HAVING (COUNT(DISTINCT ?org) > 100)

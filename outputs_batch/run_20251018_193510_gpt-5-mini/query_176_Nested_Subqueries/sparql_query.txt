PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?manufacturer ?label ?count WHERE {
  {
    SELECT (AVG(?cnt) AS ?avgCert) WHERE {
      SELECT ?m (COUNT(?cert) AS ?cnt) WHERE {
        ?m a iof-core:Manufacturer .
        OPTIONAL { ?m sdk:hasCertificate ?cert . }
      } GROUP BY ?m
    }
  }
  {
    SELECT ?manufacturer (COUNT(?cert2) AS ?count) WHERE {
      ?manufacturer a iof-core:Manufacturer .
      OPTIONAL { ?manufacturer sdk:hasCertificate ?cert2 . }
    } GROUP BY ?manufacturer
  }
  OPTIONAL { ?manufacturer rdfs:label ?label }
  FILTER(?count > ?avgCert)
}
ORDER BY DESC(?count)

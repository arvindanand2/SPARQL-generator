PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX iof-scro: <https://spec.industrialontologies.org/ontology/supplychain/SupplyChain/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?naics ?naicsCode ?naicsText ?state ?stateLabel ?count WHERE {
  {
    SELECT ?naics ?state (COUNT(DISTINCT ?m) AS ?count) WHERE {
      ?m a iof-core:Manufacturer ;
         sdk:hasPrimaryNAICSClassifier ?naics ;
         sdk:organizationLocatedIn ?site .
      ?site sdk:locatedInCity ?city .
      ?city sdk:locatedInState ?state .
    } GROUP BY ?naics ?state
  }
  OPTIONAL { ?naics sdk:hasNAICSCodeValue ?naicsCode . }
  OPTIONAL { ?naics sdk:hasNAICSTextValue ?naicsText . }
  OPTIONAL { ?state rdfs:label ?stateLabel . }

  {
    SELECT ?naics ?state ?count (COUNT(?b_state) AS ?numHigher) WHERE {
      {
        SELECT ?naics ?state (COUNT(DISTINCT ?m) AS ?count) WHERE {
          ?m a iof-core:Manufacturer ;
             sdk:hasPrimaryNAICSClassifier ?naics ;
             sdk:organizationLocatedIn ?site .
          ?site sdk:locatedInCity ?city .
          ?city sdk:locatedInState ?state .
        } GROUP BY ?naics ?state
      }
      {
        SELECT ?naics ?b_state (COUNT(DISTINCT ?m2) AS ?bcount) WHERE {
          ?m2 a iof-core:Manufacturer ;
              sdk:hasPrimaryNAICSClassifier ?naics ;
              sdk:organizationLocatedIn ?site2 .
          ?site2 sdk:locatedInCity ?city2 .
          ?city2 sdk:locatedInState ?b_state .
        } GROUP BY ?naics ?b_state
      }
      FILTER( ?bcount > ?count || (?bcount = ?count && str(?b_state) < str(?state)) )
    } GROUP BY ?naics ?state ?count
  }
  FILTER(?numHigher < 5)
}
ORDER BY ?naics DESC(?count)

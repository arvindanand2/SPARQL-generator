PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX iof-scro: <https://spec.industrialontologies.org/ontology/supplychain/SupplyChain/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT ?naics ?code ?text (COUNT(DISTINCT ?man) AS ?numManufacturers)
WHERE {
  ?man a iof-core:Manufacturer ;
       sdk:hasPrimaryNAICSClassifier ?naics .
  OPTIONAL { ?naics sdk:hasNAICSCodeValue ?code. }
  OPTIONAL { ?naics sdk:hasNAICSTextValue ?text. }
}
GROUP BY ?naics ?code ?text
ORDER BY DESC(?numManufacturers)
LIMIT 1

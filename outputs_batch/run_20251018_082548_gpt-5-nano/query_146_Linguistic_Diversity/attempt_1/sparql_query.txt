PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT DISTINCT ?company ?name
WHERE {
  ?company a iof-core:Manufacturer .
  ?company rdfs:label ?name .
  ?company sdk:hasMaterialCapability ?matCap .
  { ?matCap a <http://asu.edu/semantics/SUDOKN/CeramicProcessingCapability> . }
  UNION
  { ?matCap a <http://asu.edu/semantics/SUDOKN/CeramicMoldCastingCapability> . }
}
ORDER BY ?name

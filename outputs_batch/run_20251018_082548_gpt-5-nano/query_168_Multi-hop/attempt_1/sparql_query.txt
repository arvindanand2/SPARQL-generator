PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT DISTINCT ?manufacturer ?manufLabel ?site ?city ?cityLabel ?state ?stateLabel
WHERE {
  ?manufacturer a iof-core:Manufacturer ;
                rdfs:label ?manufLabel ;
                sdk:organizationLocatedIn ?site .
  ?site a sdk:GeospatialLocation ;
        sdk:locatedInCity ?city .
  ?city a sdk:City ;
        rdfs:label ?cityLabel ;
        sdk:locatedInState ?state .
  ?state a sdk:State ;
         rdfs:label ?stateLabel .
}

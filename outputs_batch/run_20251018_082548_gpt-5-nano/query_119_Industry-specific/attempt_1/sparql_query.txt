PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?manufacturer ?manuLabel (COUNT(DISTINCT ?ind) AS ?industryCount)
WHERE {
  ?manufacturer a iof-core:Manufacturer ;
                rdfs:label ?manuLabel ;
                sdk:suppliesToIndustry ?ind .
  ?ind rdfs:label ?indName .
  FILTER (REGEX(?indName, "Energy", "i"))
}
GROUP BY ?manufacturer ?manuLabel
ORDER BY DESC(?industryCount)
LIMIT 10

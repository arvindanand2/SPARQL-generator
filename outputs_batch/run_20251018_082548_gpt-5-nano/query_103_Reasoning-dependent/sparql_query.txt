PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?manufacturer ?name WHERE {
  ?manufacturer a iof-core:Manufacturer ;
                rdfs:label ?name ;
                sdk:hasProcessCapability ?cap .
  FILTER(?cap = <http://asu.edu/semantics/SUDOKN/PolymerDepositionCapability>)
}

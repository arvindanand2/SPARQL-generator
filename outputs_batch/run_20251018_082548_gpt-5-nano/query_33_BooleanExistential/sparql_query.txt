PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?manuf ?name (BOUND(?naics) AS ?hasPrimaryNAICS)
WHERE {
  ?manuf a iof-core:Manufacturer ;
          rdfs:label ?name .
  FILTER(CONTAINS(LCASE(?name), "pipe"))
  OPTIONAL { ?manuf sdk:hasPrimaryNAICSClassifier ?naics . }
}

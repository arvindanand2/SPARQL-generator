PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?certificate ?certificateLabel ?manufacturerCount
WHERE {
  {
    SELECT ?certificate (COUNT(DISTINCT ?manufacturer) AS ?manufacturerCount)
    WHERE {
      ?manufacturer a iof-core:Manufacturer ;
                  sdk:hasCertificate ?certificate .
    }
    GROUP BY ?certificate
  }
  OPTIONAL { ?certificate rdfs:label ?certificateLabel . }
}
ORDER BY DESC(?manufacturerCount)
LIMIT 1

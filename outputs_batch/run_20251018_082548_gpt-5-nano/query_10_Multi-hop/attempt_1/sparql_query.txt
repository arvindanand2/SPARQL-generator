PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?manufacturer ?codeValue ?codeText
WHERE {
  ?manufacturer a iof-core:Manufacturer ;
                rdfs:label ?name ;
                sdk:hasPrimaryNAICSClassifier ?naicsClassifier .
  FILTER(CONTAINS(LCASE(?name), "precision"))
  ?naicsClassifier sdk:hasNAICSCodeValue ?codeValue ;
                   sdk:hasNAICSTextValue ?codeText .
}

PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT DISTINCT ?manufacturer ?manufacturerLabel
WHERE {
  ?manufacturer a iof-core:Manufacturer .
  ?manufacturer rdfs:label ?manufacturerLabel .
  ?manufacturer sdk:suppliesToIndustry ?medInd .
  ?manufacturer sdk:suppliesToIndustry ?pharmaInd .
  ?medInd rdfs:label "MedicalDevices" .
  ?pharmaInd rdfs:label "Pharmaceuticals" .
  FILTER(?medInd != ?pharmaInd)
}
ORDER BY ?manufacturerLabel

PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?org ?name
WHERE {
  ?org a iof-core:Manufacturer ;
       rdfs:label ?name ;
       sdk:suppliesToIndustry ?indA , ?indB .
  FILTER(?indA != ?indB)
  FILTER(?indA IN (sdk:industry-001, sdk:industry-002) && ?indB IN (sdk:industry-001, sdk:industry-002))
  FILTER NOT EXISTS { ?org sdk:suppliesToIndustry ?extra .
                      FILTER(?extra NOT IN (?indA, ?indB)) }
}

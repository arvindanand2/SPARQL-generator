PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT DISTINCT ?manufacturer ?manufacturerLabel ?cityLabel ?stateLabel WHERE {
  ?manufacturer a iof-core:Manufacturer .
  OPTIONAL { ?manufacturer rdfs:label ?manufacturerLabel . }
  OPTIONAL {
    ?manufacturer sdk:organizationLocatedIn ?site .
    OPTIONAL {
      ?site sdk:locatedInCity ?city .
      OPTIONAL { ?city rdfs:label ?cityLabel . }
      OPTIONAL {
        ?city sdk:locatedInState ?stateViaCity .
        OPTIONAL { ?stateViaCity rdfs:label ?stateViaCityLabel . }
      }
    }
    OPTIONAL {
      ?site sdk:locatedInState ?stateViaSite .
      OPTIONAL { ?stateViaSite rdfs:label ?stateViaSiteLabel . }
    }
  }
  BIND(COALESCE(?stateViaCityLabel, ?stateViaSiteLabel) AS ?stateLabel)
}

PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?naics ?naicsLabel ?state ?stateLabel ?count WHERE {
  {
    SELECT ?naics ?state ?count (COUNT(?s2) AS ?better) WHERE {
      {
        SELECT ?naics ?state (COUNT(DISTINCT ?org) AS ?count) WHERE {
          ?org a iof-core:Manufacturer ;
               sdk:hasPrimaryNAICSClassifier ?naics ;
               sdk:organizationLocatedIn ?site .
          ?site sdk:locatedInCity ?city .
          ?city sdk:locatedInState ?state .
        } GROUP BY ?naics ?state
      }
      {
        SELECT ?naics ?s2 (COUNT(DISTINCT ?org2) AS ?count2) WHERE {
          ?org2 a iof-core:Manufacturer ;
                sdk:hasPrimaryNAICSClassifier ?naics ;
                sdk:organizationLocatedIn ?site2 .
          ?site2 sdk:locatedInCity ?city2 .
          ?city2 sdk:locatedInState ?s2 .
        } GROUP BY ?naics ?s2
      }
      FILTER( ?count2 > ?count || ( ?count2 = ?count && str(?s2) < str(?state) ) )
    } GROUP BY ?naics ?state ?count
  }
  FILTER( ?better < 5 )
  OPTIONAL { ?naics rdfs:label ?naicsLabel . }
  OPTIONAL { ?state rdfs:label ?stateLabel . }
}
ORDER BY ?naicsLabel DESC(?count)

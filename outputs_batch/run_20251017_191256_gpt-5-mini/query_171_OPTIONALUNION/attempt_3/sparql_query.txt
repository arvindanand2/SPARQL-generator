PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT DISTINCT ?manufacturer ?manufacturerLabel ?state ?stateLabel WHERE {
  ?manufacturer a iof-core:Manufacturer .
  OPTIONAL { ?manufacturer rdfs:label ?manufacturerLabel . }

  ?manufacturer (sdk:organizationLocatedIn/sdk:locatedInCity/sdk:locatedInState | sdk:organizationLocatedIn/sdk:locatedInState | sdk:locatedInState) ?state .

  OPTIONAL { ?state rdfs:label ?stateLabel . }
}

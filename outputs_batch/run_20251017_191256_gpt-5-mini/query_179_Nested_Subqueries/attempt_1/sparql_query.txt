PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?manufacturer ?label ?certCount WHERE {
  {
    SELECT (AVG(?c) AS ?avgCert) WHERE {
      SELECT ?m (COUNT(?cert) AS ?c) WHERE {
        ?m a iof-core:Manufacturer .
        OPTIONAL { ?m sdk:hasCertificate ?cert . }
      } GROUP BY ?m
    }
  }
  {
    SELECT ?manufacturer (COUNT(?cert) AS ?certCount) WHERE {
      ?manufacturer a iof-core:Manufacturer .
      OPTIONAL { ?manufacturer sdk:hasCertificate ?cert . }
    } GROUP BY ?manufacturer
  }
  OPTIONAL { ?manufacturer rdfs:label ?label . }
  FILTER(?certCount > ?avgCert)
}
ORDER BY DESC(?certCount)

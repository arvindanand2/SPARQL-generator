PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT DISTINCT ?org ?orgLabel ?cityLabel ?stateLabel WHERE {
  ?org a iof-core:Manufacturer .
  OPTIONAL { ?org rdfs:label ?orgLabel. }
  ?org sdk:hasNumberOfEmployees ?employeesStr .
  FILTER(xsd:integer(REPLACE(str(?employeesStr), ",", "")) > 500)
  ?org sdk:organizationLocatedIn ?site .
  ?site sdk:locatedInCity ?city .
  ?city rdfs:label ?cityLabel .
  ?city sdk:locatedInState ?state .
  ?state rdfs:label ?stateLabel .
  FILTER(str(?stateLabel) = "California")
  ?org sdk:hasProcessCapability ?cap .
  ?cap a sdk:AdditiveManufacturingCapability .
}

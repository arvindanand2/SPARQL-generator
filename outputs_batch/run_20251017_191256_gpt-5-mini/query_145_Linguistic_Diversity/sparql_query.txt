PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX iof-scro: <https://spec.industrialontologies.org/ontology/supplychain/SupplyChain/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT DISTINCT ?company ?companyLabel WHERE {
  ?company a iof-core:Manufacturer ;
           rdfs:label ?companyLabel .
  {
    ?company sdk:suppliesToIndustry ?industry .
    ?industry rdfs:label ?industryLabel .
    FILTER(regex(str(?industryLabel), "Automotive", "i"))
  }
  UNION
  {
    ?company iof-core:describedBy ?desc .
    ?desc iof-scro:hasTextValue ?descText .
    FILTER(regex(str(?descText), "service", "i"))
  }
}

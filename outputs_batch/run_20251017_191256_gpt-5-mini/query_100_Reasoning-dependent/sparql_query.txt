PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT DISTINCT ?manufacturer ?manufacturerLabel WHERE {
  ?manufacturer a iof-core:Manufacturer ;
                sdk:hasProcessCapability ?cap ;
                rdfs:label ?manufacturerLabel .
  ?cap a ?capType .
  VALUES ?capType {
    sdk:AdditiveManufacturingCapability
    sdk:FusedDepositionModelingCapability
    sdk:PowderDepositionCapability
    sdk:SolidDepositionCapability
    sdk:DropOnDemandCapability
    sdk:PolymerDepositionCapability
  }
}

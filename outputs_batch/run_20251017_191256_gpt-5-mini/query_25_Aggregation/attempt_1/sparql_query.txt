PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?certificateType (COUNT(DISTINCT ?manufacturer) AS ?manufacturerCount)
WHERE {
  ?manufacturer a iof-core:Manufacturer ;
                sdk:hasCertificate ?certificateInstance .
  ?certificateInstance a ?certificateType .
}
GROUP BY ?certificateType
ORDER BY DESC(?manufacturerCount)

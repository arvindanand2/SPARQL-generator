PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT DISTINCT ?cert ?certLabel WHERE {
  ?man a iof-core:Manufacturer ;
       rdfs:label ?name ;
       sdk:hasCertificate ?cert .
  FILTER(CONTAINS(LCASE(STR(?name)), "plastics"))
  OPTIONAL { ?cert rdfs:label ?certLabel . }
}

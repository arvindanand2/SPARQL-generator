PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT DISTINCT ?manufacturer ?cityState WHERE {
  ?manufacturer a iof-core:Manufacturer .
  OPTIONAL {
    ?manufacturer sdk:organizationLocatedIn ?site .
    ?site sdk:locatedInCity ?city .
    ?city rdfs:label ?cityLabelRaw .
    OPTIONAL {
      ?city sdk:locatedInState ?state .
      ?state rdfs:label ?stateLabelRaw .
    }
  }
  BIND(COALESCE(?cityLabelRaw, "Unknown") AS ?cityLabel)
  BIND(COALESCE(?stateLabelRaw, "Unknown") AS ?stateLabel)
  BIND(CONCAT(?cityLabel, ", ", ?stateLabel) AS ?cityState)
}
LIMIT 10000

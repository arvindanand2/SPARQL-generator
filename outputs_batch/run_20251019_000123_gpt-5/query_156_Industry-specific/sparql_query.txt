PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?org ?orgLabel
WHERE {
  ?org a iof-core:Manufacturer ;
       sdk:suppliesToIndustry ?ind ;
       sdk:suppliesToIndustry ?renInd ;
       sdk:suppliesToIndustry ?utilInd .
  ?renInd a sdk:RenewableEnergyIndustry .
  ?utilInd a sdk:UtilitiesIndustry .
  OPTIONAL { ?org rdfs:label ?orgLabel. }
}
GROUP BY ?org ?orgLabel
HAVING (COUNT(DISTINCT ?ind) = 2)

PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT DISTINCT ?manufacturer ?manufacturerLabel ?lat
WHERE {
  ?manufacturer a iof-core:Manufacturer ;
                rdfs:label ?manufacturerLabel ;
                sdk:suppliesToIndustry ?indAgri ;
                sdk:suppliesToIndustry ?indConstr ;
                sdk:organizationLocatedIn ?site .
  ?indAgri rdfs:label ?agriLabel .
  ?indConstr rdfs:label ?constrLabel .
  FILTER(LCASE(STR(?agriLabel)) = "agriculture")
  FILTER(LCASE(STR(?constrLabel)) = "construction")
  ?site sdk:hasSpatialCoordinates ?coords .
  ?coords sdk:hasLatitudeValue ?lat .
  FILTER(xsd:float(?lat) > 40.0)
}

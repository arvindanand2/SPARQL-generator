PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT DISTINCT ?manufacturer ?manufacturerLabel
WHERE {
  ?manufacturer a iof-core:Manufacturer ;
                rdfs:label ?manufacturerLabel .

  FILTER EXISTS {
    ?manufacturer sdk:hasMaterialCapability ?capAllowed .
    ?capAllowed a ?typeAllowed .
    FILTER(?typeAllowed IN (
      sdk:UrethaneProcessingCapability,
      sdk:StainlessSteelProcessingCapability,
      sdk:BronzeProcessingCapability
    ))
  }

  FILTER NOT EXISTS {
    ?manufacturer sdk:hasMaterialCapability ?capOther .
    ?capOther a ?typeOther .
    FILTER(?typeOther NOT IN (
      sdk:UrethaneProcessingCapability,
      sdk:StainlessSteelProcessingCapability,
      sdk:BronzeProcessingCapability
    ))
  }
}

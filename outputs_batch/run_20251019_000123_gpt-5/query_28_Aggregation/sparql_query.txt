PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?manufacturer ?manufacturerLabel (COUNT(DISTINCT ?capability) AS ?totalCapabilities)
WHERE {
  ?manufacturer a iof-core:Manufacturer .
  OPTIONAL { ?manufacturer rdfs:label ?manufacturerLabel . }
  {
    ?manufacturer sdk:hasProcessCapability ?capability .
  } UNION {
    ?manufacturer sdk:hasMaterialCapability ?capability .
  }
}
GROUP BY ?manufacturer ?manufacturerLabel

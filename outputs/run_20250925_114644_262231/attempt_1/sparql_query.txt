PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT DISTINCT ?manufacturer ?label ?latitude ?longitude
WHERE {
  ?manufacturer a iof-core:Manufacturer ;
                rdfs:label ?label ;
                sdk:organizationLocatedIn ?site .
  ?site sdk:hasSpatialCoordinates ?coords .
  ?coords sdk:hasLatitudeValue ?latitude ;
          sdk:hasLongitudeValue ?longitude .
  # Must be in the Midwest (assume Midwest states are known as sdk:state-XXX, list omitted here)
  ?site sdk:locatedInCity ?city .
  ?city sdk:locatedInState ?state .
  ?state rdfs:label ?stateLabel .
  FILTER (?stateLabel IN ("Illinois", "Indiana", "Iowa", "Kansas", "Michigan", "Minnesota", "Missouri", "Nebraska", "North Dakota", "Ohio", "South Dakota", "Wisconsin"))
  # Must supply to Aerospace or Agriculture industry
  ?manufacturer sdk:suppliesToIndustry ?industry .
  ?industry rdfs:label ?industryLabel .
  FILTER (lcase(?industryLabel) = "aerospace" || lcase(?industryLabel) = "agriculture")
  # Get distance to ASU Tempe campus (lat: 33.4212, long: -111.9336)
  BIND(xsd:double(?latitude) AS ?lat)
  BIND(xsd:double(?longitude) AS ?long)
  BIND(33.4212 AS ?asuLat)
  BIND(-111.9336 AS ?asuLong)
  BIND(
    6371 * 2 * ASIN(
      SQRT(
        SIN(RADIANS((?lat - ?asuLat)/2)) * SIN(RADIANS((?lat - ?asuLat)/2)) +
        COS(RADIANS(?asuLat)) * COS(RADIANS(?lat)) *
        SIN(RADIANS((?long - ?asuLong)/2)) * SIN(RADIANS((?long - ?asuLong)/2))
      )
    )
    AS ?distanceKm
  )
}
ORDER BY ?distanceKm

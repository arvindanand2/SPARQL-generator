PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT DISTINCT ?manufacturer ?manufacturerLabel ?distance
WHERE {
  ?manufacturer a iof-core:Manufacturer ;
                rdfs:label ?manufacturerLabel ;
                sdk:organizationLocatedIn ?site ;
                sdk:hasCertificate ?cert .
  ?site sdk:hasSpatialCoordinates ?coords .
  ?coords sdk:hasLatitudeValue ?lat ;
          sdk:hasLongitudeValue ?long .
  FILTER (
    ?cert IN (
      <http://asu.edu/semantics/SUDOKN/ISO9001>,
      <http://asu.edu/semantics/SUDOKN/AS9100Certificate>,
      <http://asu.edu/semantics/SUDOKN/AS9120>,
      <http://asu.edu/semantics/SUDOKN/AS9003>,
      <http://asu.edu/semantics/SUDOKN/AS9000>,
      <http://asu.edu/semantics/SUDOKN/ISOQualityCertificate>,
      <http://asu.edu/semantics/SUDOKN/QualityCertificate>,
      <http://asu.edu/semantics/SUDOKN/AerospaceQualityCertificate>,
      <http://asu.edu/semantics/SUDOKN/QSQuality>
    )
  )
  # Midwest filter using city/state label matching (example states)
  ?site sdk:locatedInCity ?city .
  ?city sdk:locatedInState ?state .
  ?state rdfs:label ?stateLabel .
  FILTER (
    ?stateLabel IN (
      "Illinois", "Indiana", "Iowa", "Kansas", "Michigan",
      "Minnesota", "Missouri", "Nebraska", "North Dakota",
      "Ohio", "South Dakota", "Wisconsin"
    )
  )
  # Coordinates for ASU Tempe campus
  BIND("33.421204"^^xsd:double AS ?asuLat)
  BIND("-111.93301"^^xsd:double AS ?asuLong)
  # Convert to float
  BIND(xsd:double(?lat) AS ?mLat)
  BIND(xsd:double(?long) AS ?mLong)
  # Haversine distance (km)
  BIND(0.01745329252 AS ?radPerDeg)
  BIND((?asuLat-?mLat)*?radPerDeg AS ?dLat)
  BIND((?asuLong-?mLong)*?radPerDeg AS ?dLng)
  BIND(sin(?dLat/2)*sin(?dLat/2) + cos(?mLat*?radPerDeg)*cos(?asuLat*?radPerDeg)*sin(?dLng/2)*sin(?dLng/2) AS ?a)
  BIND(2*atan2(sqrt(?a), sqrt(1-?a)) AS ?c)
  BIND(6371.0*?c AS ?distance) # 6371 km = earth radius
}
ORDER BY ?distance

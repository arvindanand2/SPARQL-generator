PREFIX sdk: <http://asu.edu/semantics/SUDOKN/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
PREFIX iof-core: <https://spec.industrialontologies.org/ontology/core/Core/>

SELECT DISTINCT ?manufacturer ?manufacturerLabel
WHERE {
  ?manufacturer a iof-core:Manufacturer ;
                rdfs:label ?manufacturerLabel ;
                sdk:suppliesToIndustry ?industry ;
                sdk:organizationLocatedIn ?site .
  ?industry rdfs:label ?industryLabel .
  ?site sdk:locatedInCity ?city .
  ?city sdk:locatedInState ?state .
  ?state rdfs:label ?stateLabel .
  FILTER(CONTAINS(LCASE(?industryLabel), "aerospace"))
  FILTER(?stateLabel IN ("Illinois", "Indiana", "Iowa", "Kansas", "Michigan", "Minnesota", "Missouri", "Nebraska", "North Dakota", "Ohio", "South Dakota", "Wisconsin"))
}
